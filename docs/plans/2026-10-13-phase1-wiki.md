# 1단계: LLM 위키 핵심 문서 구현 계획

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 진단 알고리즘과 팔레트의 근거가 되는 LLM 위키(`raw/`, `wiki/`, `CLAUDE.md`)를 만들고, 앱이 읽을 데이터 초안(`app/data/seasons.json`, `app/data/palettes.json`)을 위키에서 옮겨 적는다.

**Architecture:** `raw/`는 수집한 원본 자료 요약(수정 금지), `wiki/`는 주제별 정리 문서, `app/data/*.json`은 위키에서 옮긴 숫자·색상. 위키 린터(`tools/wiki-lint.js`)와 데이터 검증기(`tools/validate-data.js`)가 링크·출처·형식을 자동으로 검사한다.

**Tech Stack:** Markdown, JSON, Node 24 (내장 `node:test`, 외부 패키지 없음)

**Spec:** `docs/specs/2026-10-12-personal-color-booth-design.md`

## Global Constraints

- 모든 문서는 한국어로 쓴다. 코드 식별자·JSON 키는 영어.
- 위키의 모든 숫자 기준값은 `status: 가설`로 표시한다 (스펙 6장: "모든 숫자 기준값은 가설").
- `raw/` 파일은 만든 뒤 수정하지 않는다. 고칠 내용은 `wiki/`에 쓴다.
- 위키를 바꾸면 `wiki/log.md`에 한 줄 기록한다.
- 외부 npm 패키지를 설치하지 않는다 (스펙 8장: 빌드 도구 없음).
- 4계절 id 순서는 항상 `spring, summer, autumn, winter`.
- 12톤 id: `spring-light, spring-bright, spring-true, summer-light, summer-soft, summer-true, autumn-soft, autumn-deep, autumn-true, winter-bright, winter-deep, winter-true`.
- 4축 id 순서는 항상 `warmth, value, chroma, contrast`. 점수 범위 0~100.
- 색 코드는 대문자 6자리 `#RRGGBB`.
- 연구 전용 데이터셋(Deep Armocromia, HF armocromia), AI Hub, Colorinsight, 외부 진단 API는 쓰지 않는다 (스펙 12장).

## Review Focus

1. 위키 문서끼리 숫자가 어긋남 (예: `scoring/`의 임계값과 `seasons.json` 값이 다름) → 검증기가 JSON 내부 일관성을 잡고, Task 7·8 Step에서 위키 표와 JSON을 나란히 대조한다.

function makeRepo(files) {
  const root = mkdtempSync(join(tmpdir(), 'wiki-'));
  for (const [path, content] of Object.entries(files)) {

- [ ] **Step 4: 구현** — `tools/wiki-lint.js`

```js
import { readdirSync, readFileSync, statSync, existsSync } from 'node:fs';
import { join, dirname, resolve, relative } from 'node:path';
import { pathToFileURL } from 'node:url';

const STATUSES = new Set(['가설', '확정', '참고']);
const SPECIAL = new Set(['index.md', 'log.md']);

const toPosix = (p) => p.replaceAll('\\', '/');

function listMarkdown(dir) {
  const out = [];
  for (const name of readdirSync(dir)) {
    const full = join(dir, name);
    if (statSync(full).isDirectory()) out.push(...listMarkdown(full));
    else if (name.endsWith('.md')) out.push(full);
  }
  return out;
}

export function parseFrontMatter(text) {
  const match = text.match(/^---\r?\n([\s\S]*?)\r?\n---/);
  if (!match) return null;
  const data = {};
  let key = null;
  for (const line of match[1].split(/\r?\n/)) {
    const kv = line.match(/^(\w+):\s*(.*)$/);
    if (kv) {
      key = kv[1];
      data[key] = kv[2].trim() === '' ? [] : kv[2].trim();
      continue;
    }
    const item = line.match(/^\s*-\s+(.+)$/);
    if (item && key && Array.isArray(data[key])) data[key].push(item[1].trim());
  }
  return data;
}

export function lintWiki(wikiDir) {
  const errors = [];
  const repoRoot = resolve(wikiDir, '..');
  const files = listMarkdown(wikiDir);
  if (!existsSync(join(wikiDir, 'index.md'))) errors.push('index.md 없음');
  const linkedFromIndex = new Set();

  for (const file of files) {
    const rel = toPosix(relative(wikiDir, file));
    const text = readFileSync(file, 'utf8');

    if (!SPECIAL.has(rel)) {
      const fm = parseFrontMatter(text);
      if (!fm) {
        errors.push(`${rel}: front matter 없음`);
      } else {
        if (!Array.isArray(fm.sources) || fm.sources.length === 0) {
          errors.push(`${rel}: sources 비어 있음`);
        } else {
          for (const src of fm.sources) {
            if (/^https?:\/\//.test(src)) continue;
            if (!existsSync(resolve(repoRoot, src))) errors.push(`${rel}: 출처 파일 없음 ${src}`);
          }
        }
        if (!STATUSES.has(fm.status)) errors.push(`${rel}: status는 가설/확정/참고 중 하나여야 함`);
      }
    }

    for (const m of text.matchAll(/\]\(([^)#\s]+\.md)(#[^)]*)?\)/g)) {
      const target = m[1];
      if (/^https?:\/\//.test(target)) continue;
      const abs = resolve(dirname(file), target);
      if (!existsSync(abs)) errors.push(`${rel}: 깨진 링크 ${target}`);
      else if (rel === 'index.md') linkedFromIndex.add(toPosix(relative(wikiDir, abs)));
    }
  }

  for (const file of files) {
    const rel = toPosix(relative(wikiDir, file));
    if (rel !== 'index.md' && !linkedFromIndex.has(rel)) {
      errors.push(`${rel}: index.md에서 링크되지 않음`);
    }
  }
  return errors;
}

if (process.argv[1] && import.meta.url === pathToFileURL(process.argv[1]).href) {
  const errors = lintWiki(process.argv[2] ?? 'wiki');
  for (const e of errors) console.error(e);
  console.log(errors.length === 0 ? '위키 검사 통과' : `오류 ${errors.length}개`);
  process.exit(errors.length === 0 ? 0 : 1);
}
```

import { readFileSync } from 'node:fs';
import { pathToFileURL } from 'node:url';

export const SEASONS = ['spring', 'summer', 'autumn', 'winter'];
export const AXES = ['warmth', 'value', 'chroma', 'contrast'];
export const TONES = [
  'spring-light', 'spring-bright', 'spring-true',
  'summer-light', 'summer-soft', 'summer-true',
  'autumn-soft', 'autumn-deep', 'autumn-true',
  'winter-bright', 'winter-deep', 'winter-true',
];
const THRESHOLD_KEYS = [
  'borderlineGapPct', 'minFaceWidthRatio', 'maxYawDeg', 'holdSeconds', 'framesPerCapture',
  'trimPct', 'minSkinL', 'scleraMaxChroma', 'hairDyeMinChroma', 'lipProductMinChroma', 'idleResetSeconds',
];
const HEX = /^#[0-9A-F]{6}$/;
const isNum = (v) => typeof v === 'number' && Number.isFinite(v);

export function validateSeasons(s) {
  const errors = [];
  for (const axis of AXES) {
    const mapping = s.axisMapping?.[axis];
    const weights = s.featureWeights?.[axis];
    if (!mapping || !weights) { errors.push(`${axis}: axisMapping/featureWeights 없음`); continue; }
    for (const [key, range] of Object.entries(mapping)) {
      if (!Array.isArray(range) || range.length !== 2 || !isNum(range[0]) || !isNum(range[1]) || range[0] >= range[1]) {
        errors.push(`${axis}.${key}: 범위는 [min, max]이고 min < max여야 함`);
      }
    }
    let sum = 0;
    for (const [key, w] of Object.entries(weights)) {
      if (!(key in mapping)) errors.push(`${axis}.${key}: axisMapping에 없는 가중치 키`);
      if (!isNum(w) || w < 0) errors.push(`${axis}.${key}: 가중치는 0 이상 숫자`);
      sum += isNum(w) ? w : 0;
    }
    if (Math.abs(sum - 1) > 0.001) errors.push(`${axis}: 가중치 합이 1이 아님 (${sum})`);
  }
  for (const season of SEASONS) {
    const p = s.seasons?.[season]?.prototype;
    if (!Array.isArray(p) || p.length !== 4 || !p.every((v) => isNum(v) && v >= 0 && v <= 100)) {
      errors.push(`${season}: prototype은 0~100 숫자 4개`);
    }
  }
  if (!Array.isArray(s.axisWeights) || s.axisWeights.length !== 4 || !s.axisWeights.every((v) => isNum(v) && v > 0)) {
    errors.push('axisWeights: 0보다 큰 숫자 4개');
  }
  for (const tone of TONES) {
    const rule = s.toneRules?.[tone];
    if (!rule) { errors.push(`${tone}: toneRules 없음`); continue; }
    if (rule.season !== tone.split('-')[0]) errors.push(`${tone}: season이 id와 다름`);
    if (!AXES.includes(rule.axis)) errors.push(`${tone}: axis가 잘못됨`);
    if (rule.direction !== 1 && rule.direction !== -1) errors.push(`${tone}: direction은 1 또는 -1`);
  }
  for (const key of THRESHOLD_KEYS) {
    if (!isNum(s.thresholds?.[key])) errors.push(`thresholds.${key}: 숫자 없음`);
  }
  return errors;
}

function checkColors(list, label, errors, exactCount) {
  if (!Array.isArray(list) || list.length === 0) { errors.push(`${label}: 색 목록이 비어 있음`); return; }
  if (exactCount !== undefined && list.length !== exactCount) errors.push(`${label}: 정확히 ${exactCount}개여야 함 (${list.length}개)`);
  for (const item of list) {
    if (typeof item?.name !== 'string' || item.name === '') errors.push(`${label}: 색 이름 없음`);
    if (!HEX.test(item?.hex ?? '')) errors.push(`${label}: 잘못된 hex ${item?.hex}`);
  }
}

export function validatePalettes(p) {
  const errors = [];
  for (const tone of TONES) {
    const t = p.tones?.[tone];
    if (!t) { errors.push(`${tone}: 팔레트 없음`); continue; }
    if (typeof t.name !== 'string' || t.name === '') errors.push(`${tone}: name 없음`);
    if (!Array.isArray(t.pccsTones) || t.pccsTones.length === 0) errors.push(`${tone}: pccsTones 없음`);
    checkColors(t.best, `${tone}.best`, errors, 6);
    checkColors(t.worst, `${tone}.worst`, errors, 4);
    const best = new Set((t.best ?? []).map((c) => c.hex));
    for (const c of t.worst ?? []) if (best.has(c.hex)) errors.push(`${tone}: ${c.hex}가 best와 worst에 모두 있음`);
  }
  for (const season of SEASONS) {
    const s = p.seasons?.[season];
    if (!s) { errors.push(`${season}: 계절 추가 항목 없음`); continue; }
    for (const part of ['lip', 'blush', 'shadow']) checkColors(s.makeup?.[part], `${season}.makeup.${part}`, errors);
    for (const part of ['tops', 'glasses', 'hair']) checkColors(s[part], `${season}.${part}`, errors);
    if (!Array.isArray(s.metals) || s.metals.length === 0) errors.push(`${season}.metals: 비어 있음`);
    if (!Array.isArray(s.keywords) || s.keywords.length !== 3) errors.push(`${season}.keywords: 정확히 3개여야 함`);
  }
  return errors;
}

export function hexToHueDeg(hex) {
  const lin = (v) => { v /= 255; return v <= 0.04045 ? v / 12.92 : ((v + 0.055) / 1.055) ** 2.4; };
  const [r, g, b] = [1, 3, 5].map((i) => lin(parseInt(hex.slice(i, i + 2), 16)));
  const x = (0.4124564 * r + 0.3575761 * g + 0.1804375 * b) / 0.95047;
  const y = 0.2126729 * r + 0.7151522 * g + 0.0721750 * b;
  const z = (0.0193339 * r + 0.1191920 * g + 0.9503041 * b) / 1.08883;
  const f = (t) => (t > 216 / 24389 ? Math.cbrt(t) : (24389 / 27 * t + 16) / 116);
  const a = 500 * (f(x) - f(y));
  const bb = 200 * (f(y) - f(z));
  return (Math.atan2(bb, a) * 180 / Math.PI + 360) % 360;
}

if (process.argv[1] && import.meta.url === pathToFileURL(process.argv[1]).href) {
  const [seasonsPath, palettesPath] = process.argv.slice(2);
  const errors = [
    ...validateSeasons(JSON.parse(readFileSync(seasonsPath, 'utf8'))).map((e) => `seasons: ${e}`),
    ...validatePalettes(JSON.parse(readFileSync(palettesPath, 'utf8'))).map((e) => `palettes: ${e}`),
  ];
  for (const e of errors) console.error(e);
  console.log(errors.length === 0 ? '데이터 검사 통과' : `오류 ${errors.length}개`);
  process.exit(errors.length === 0 ? 0 : 1);
}

```js
import { test } from 'node:test';
import assert from 'node:assert/strict';
import { readFileSync } from 'node:fs';
import { TONES, validatePalettes, hexToHueDeg } from '../tools/validate-data.js';

const palettes = JSON.parse(readFileSync(new URL('../app/data/palettes.json', import.meta.url), 'utf8'));

// 무채색(검정·흰색·회색)은 색상각이 의미 없으므로 제외
function isNeutral(hex) {
  const [r, g, b] = [1, 3, 5].map((i) => parseInt(hex.slice(i, i + 2), 16));
  return Math.max(r, g, b) - Math.min(r, g, b) < 20;
}

// 색상각 30~100°(주황~노랑 쪽)에 있는 유채색의 비율
function warmShare(colors) {
  const chromatic = colors.filter((c) => !isNeutral(c.hex));
  const warm = chromatic.filter((c) => { const h = hexToHueDeg(c.hex); return h >= 30 && h <= 100; });
  return chromatic.length === 0 ? 0 : warm.length / chromatic.length;
}

test('palettes.json은 형식 검사를 통과한다', () => {
  assert.deepEqual(validatePalettes(palettes), []);
});

test('웜 톤(봄·가을) 베스트는 노란 쪽 색이 절반 이상', () => {
  for (const tone of TONES.filter((t) => t.startsWith('spring') || t.startsWith('autumn'))) {
    assert.ok(warmShare(palettes.tones[tone].best) >= 0.5, `${tone} 베스트에 노란 쪽 색이 부족함`);
  }
});

test('쿨 톤(여름·겨울) 베스트는 노란 쪽 색이 1/3 이하', () => {
  for (const tone of TONES.filter((t) => t.startsWith('summer') || t.startsWith('winter'))) {
    assert.ok(warmShare(palettes.tones[tone].best) <= 1 / 3, `${tone} 베스트에 노란 쪽 색이 많음`);
  }
});
```
